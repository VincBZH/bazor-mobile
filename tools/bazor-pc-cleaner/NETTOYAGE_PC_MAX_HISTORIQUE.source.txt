$Stamp = Get-Date -Format 'yyyyMMdd_HHmmss'
$LogFile = Join-Path $LogFolder ("NETTOYAGE_PC_MAX_{0}.log" -f $Stamp)
function Log([string]$s) {
    $line = (Get-Date -Format 'HH:mm:ss') + ' ' + $s
    Write-Host $line
    try { Add-Content -LiteralPath $LogFile -Value $line -Encoding UTF8 -ErrorAction Stop } catch {}
}
function FreeGB {
    try { return [math]::Round((Get-CimInstance Win32_LogicalDisk -Filter "DeviceID='C:'").FreeSpace / 1GB, 2) }
    catch { try { return [math]::Round((Get-PSDrive C).Free/1GB,2) } catch { return -1 } }
}
function Clear-Contents([string]$Folder, [string]$Label) {
    if (![string]::IsNullOrWhiteSpace($Folder) -and (Test-Path -LiteralPath $Folder -PathType Container)) {
        $b = FreeGB
        Log ("TRAITEMENT : {0} -> {1}" -f $Label,$Folder)
        Get-ChildItem -LiteralPath $Folder -Force -ErrorAction SilentlyContinue | ForEach-Object {
            try { Remove-Item -LiteralPath $_.FullName -Recurse -Force -ErrorAction Stop }
            catch { } # fichiers en cours d'utilisation: laisses en place
        }
        Log ("GAIN observe: {0} Go" -f [math]::Round((FreeGB)-$b,2))
    }
}
function Clear-OldFiles([string]$Folder, [int]$Days, [string]$Label) {
    if (!(Test-Path -LiteralPath $Folder -PathType Container)) { return }
    $b=FreeGB
    Log ("FICHIERS DE PLUS DE {0} JOURS : {1}" -f $Days,$Label)
    $limit = (Get-Date).AddDays(-$Days)
    Get-ChildItem -LiteralPath $Folder -File -Recurse -Force -ErrorAction SilentlyContinue | Where-Object {
        $_.LastWriteTime -lt $limit -and $_.Name -notmatch '(?i)mcafee|mc-fw-host'
    } | ForEach-Object { try { Remove-Item -LiteralPath $_.FullName -Force -ErrorAction Stop } catch {} }
    Log ("GAIN observe: {0} Go" -f [math]::Round((FreeGB)-$b,2))
}
function FolderGB([string]$Folder) {
    if (!(Test-Path -LiteralPath $Folder)) { return 0 }
    try {
        $total = (Get-ChildItem -LiteralPath $Folder -Recurse -File -Force -ErrorAction SilentlyContinue | Measure-Object -Property Length -Sum).Sum
        return [math]::Round($total/1GB,2)
    } catch { return -1 }
}
